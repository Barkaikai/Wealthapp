import React, { useState } from "react";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";

export default function WealthForgeDashboard() {
  const [portfolio, setPortfolio] = useState(null);
  const [aiSuggestions, setAiSuggestions] = useState([]);
  const [briefing, setBriefing] = useState("");

  const fetchPortfolio = async () => {
    const res = await fetch("/api/portfolio");
    const data = await res.json();
    setPortfolio(data);
  };

  const generateSuggestions = async () => {
    const res = await fetch("/api/trade-suggestions");
    const data = await res.json();
    setAiSuggestions(data);
  };

  const generateBriefing = async () => {
    const res = await fetch("/api/ai-briefing");
    const data = await res.json();
    setBriefing(data.message);
  };

  const handleReceiptUpload = async (event) => {
    const file = event.target.files[0];
    if (!file) return;

    const formData = new FormData();
    formData.append("receipt", file);

    const res = await fetch("/api/analyze-receipt", {
      method: "POST",
      body: formData,
    });
    const data = await res.json();
    alert(`AI Analysis: ${data.summary}`);
  };

  return (
    <div className="min-h-screen bg-gradient-to-b from-[#f9fbfd] to-[#e7efff] text-gray-900 p-6">
      <h1 className="text-4xl font-bold text-[#0052ff] mb-4">💰 WealthForge Dashboard</h1>

      {/* PORTFOLIO */}
      <Card className="mb-6 shadow-xl rounded-2xl">
        <CardContent>
          <h2 className="text-2xl font-semibold mb-2">Portfolio Overview</h2>
          <Button onClick={fetchPortfolio}>Refresh Portfolio</Button>
          {portfolio && (
            <div className="mt-4">
              <p>Total Value: ${portfolio.total_value.toFixed(2)}</p>
              <p>Daily Profit: ${portfolio.daily_profit.toFixed(2)}</p>
              <p>Goal Progress: {portfolio.progress.toFixed(1)}%</p>
            </div>
          )}
        </CardContent>
      </Card>

      {/* AI TRADE SUGGESTIONS */}
      <Card className="mb-6 shadow-xl rounded-2xl">
        <CardContent>
          <h2 className="text-2xl font-semibold mb-2">AI Trade Suggestions</h2>
          <Button onClick={generateSuggestions}>Generate</Button>
          <div className="mt-4">
            {aiSuggestions.map((s, i) => (
              <p key={i}>{s}</p>
            ))}
          </div>
        </CardContent>
      </Card>

      {/* RECEIPT ANALYZER */}
      <Card className="mb-6 shadow-xl rounded-2xl">
        <CardContent>
          <h2 className="text-2xl font-semibold mb-2">📷 Receipt Analyzer</h2>
          <input type="file" accept="image/*" onChange={handleReceiptUpload} />
        </CardContent>
      </Card>

      {/* CALENDAR SYNC */}
      <Card className="mb-6 shadow-xl rounded-2xl">
        <CardContent>
          <h2 className="text-2xl font-semibold mb-2">🗓️ Calendar Sync</h2>
          <Button onClick={() => alert("Syncing with Gmail... (setup backend)")}>Sync Gmail Calendar</Button>
        </CardContent>
      </Card>

      {/* AI BRIEFING */}
      <Card className="shadow-xl rounded-2xl">
        <CardContent>
          <h2 className="text-2xl font-semibold mb-2">🤖 AI Briefing</h2>
          <Button onClick={generateBriefing}>Generate AI Briefing</Button>
          {briefing && <p className="mt-4">{briefing}</p>}
        </CardContent>
      </Card>
    </div>
  );
}

from fastapi import FastAPI, UploadFile, Form
from fastapi.middleware.cors import CORSMiddleware
import pandas as pd
import io, json

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Load portfolio from CSV
@app.get("/api/portfolio")
def get_portfolio():
    df = pd.read_csv("portfolio_export_2025-10-24.csv")
    df.columns = [c.strip().lower() for c in df.columns]
    total_value = df["value"].sum()
    avg_change = df["24h"].astype(str).str.replace("%", "").astype(float).mean()
    daily_profit = total_value * (avg_change / 100)
    goal = 100
    progress = min(daily_profit / goal * 100, 100)
    return {
        "total_value": total_value,
        "daily_profit": daily_profit,
        "progress": progress
    }
