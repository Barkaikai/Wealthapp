TypeScript Example (server.ts)
import express, { Request, Response } from "express";
import cors from "cors";

const app = express();
app.use(cors());
app.use(express.json());

// ✅ Health check route
app.get("/", (_req: Request, res: Response) => {
  res.json({ message: "✅ Node backend running smoothly on Replit!" });
});

// ✅ Example route: Portfolio
app.get("/api/portfolio", async (_req: Request, res: Response) => {
  try {
    // Fake data — replace with your real portfolio logic
    const totalValue = 5000;
    const dailyProfit = 120;
    res.json({
      totalValue,
      dailyProfit,
      progress: (dailyProfit / 100) * 100,
      message: "Portfolio data loaded successfully",
    });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "Server error" });
  }
});

// ✅ Prevent hanging routes (timeout guard)
app.use((req: Request, res: Response) => {
  res.status(404).json({ error: "Not found" });
});

// ✅ Launch server
const PORT = process.env.PORT || 3000;
app.listen(PORT, "0.0.0.0", () => {
  console.log(`🚀 Server running on port ${PORT}`);
});

package.json Fix

Ensure this section exists in your package.json:

{
  "type": "module",
  "scripts": {
    "start": "node server.js"
  },
  "dependencies": {
    "express": "^4.18.2",
    "cors": "^2.8.5"
  }
}


If you’re using TypeScript:

"scripts": {
  "start": "ts-node server.ts"
},
"devDependencies": {
  "typescript": "^5.3.0",
  "ts-node": "^10.9.1"
}

Replit .replit file

Add this line:

run = "npm start"


Or if you’re using TypeScript:

run = "npx ts-node server.ts"

⚙️ Debug Checklist (Step-by-Step)

In Replit, open your project

Go to package.json

Make sure "type": "module" is there

Open your server.ts (or index.js)

Copy the code above

Click Run

You should see in the Replit console:

🚀 Server running on port 3000


Visit your Replit web URL — it should show:

{"message": "✅ Node backend running smoothly on Replit!"}

🩺 Optional — Add Auto Health Monitor (Prevents Future Crashes)

Add this snippet to your server:

setInterval(() => {
  fetch(`http://localhost:${PORT}/api/portfolio`).catch(() => {
    console.log("⛑️ Restarting server due to hang...");
    process.exit(1);
  });
}, 5 * 60 * 1000); // check every 5 min


That auto-restarts if it hangs.
